cmake_minimum_required(VERSION 3.20)
project(trellis_hello LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_AUTOMOC ON)
find_package(ECM REQUIRED NO_MODULE)
set(CMAKE_MODULE_PATH ${ECM_MODULE_PATH})
include(KDEInstallDirs)
include(KDECMakeSettings)
find_package(Qt6 REQUIRED COMPONENTS Core Gui Widgets Quick DBus)
find_package(KWin REQUIRED)
find_package(KF6CoreAddons REQUIRED)
add_library(trellis_hello MODULE hello.cpp)
target_link_libraries(trellis_hello PRIVATE KWin::kwin KF6::CoreAddons)
set_target_properties(trellis_hello PROPERTIES PREFIX "" OUTPUT_NAME trellis_hello)
